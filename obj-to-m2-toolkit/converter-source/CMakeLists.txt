cmake_minimum_required(VERSION 3.20)
project(OBJtoM2 LANGUAGES CXX)
add_executable(OBJtoM2 main.cpp m2.cpp)
target_compile_features(OBJtoM2 PRIVATE cxx_std_11)
if(MSVC)
  target_compile_definitions(OBJtoM2 PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(OBJtoM2 PRIVATE /EHsc)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME obj_roundtrip COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_conversion.py" "$<TARGET_FILE:OBJtoM2>")
  endif()
endif()
